% Adaptive, semantic layout profiles for the resume template.
%
% Public entry points:
%   \ResumeDensity       airy | balanced | compact | dense
%   \ResumeContentFile   content file selected by resume.tex
%   \ResumePageFill      natural | elastic (fit-tool controlled)
%
% Components should consume the public lengths and helpers below instead of
% embedding profile-specific dimensions.  The defaults intentionally keep
% every stretch component finite: \flushbottom can distribute a modest amount
% of spare height, while the fit tool moves to a more suitable profile rather
% than relying on \vfill or negative spacing.

\providecommand{\ResumeDensity}{balanced}
\providecommand{\ResumeContentFile}{content.tex}
\providecommand{\ResumePageFill}{natural}

\ifdefstring{\ResumeDensity}{airy}{%
  \ResumeUseAiryDensity
}{\ifdefstring{\ResumeDensity}{balanced}{%
  \ResumeUseBalancedDensity
}{\ifdefstring{\ResumeDensity}{compact}{%
  \ResumeUseCompactDensity
}{\ifdefstring{\ResumeDensity}{dense}{%
  \ResumeUseDenseDensity
}{%
  \PackageError{resume-layout}
    {Unknown density profile '\ResumeDensity'}
    {Use airy, balanced, compact, or dense. Falling back to balanced.}%
  \renewcommand{\ResumeDensity}{balanced}%
  \ResumeUseBalancedDensity
}}}}

\ifdefstring{\ResumePageFill}{natural}{%
  \ResumeElasticPageFillfalse
}{\ifdefstring{\ResumePageFill}{elastic}{%
  \ifdefstring{\ResumeDensity}{balanced}{%
    \ResumeElasticPageFilltrue
    \ResumeFlushBottomtrue
    % Finite-order stretch only. The fit tool enables this overlay after the
    % natural balanced render proves that the exact page count is underfilled.
    \setlength{\ResumeSectionBefore}{3.25mm plus 4.50mm minus 0.40mm}%
    \setlength{\ResumeSectionAfter}{1.55mm plus 4.50mm minus 0.24mm}%
    \setlength{\ResumeListItemSep}{0.50mm plus 3.25mm minus 0.09mm}%
    \setlength{\ResumeListTopSep}{0.52mm plus 3.25mm minus 0.09mm}%
    \setlength{\ResumeHeaderAfter}{2.15mm plus 4.50mm minus 0.16mm}%
    \setlength{\ResumeEntryBefore}{2.15mm plus 4.50mm minus 0.15mm}%
    \setlength{\ResumeEntryAfter}{1.25mm plus 4.50mm minus 0.12mm}%
  }{%
    \PackageError{resume-layout}
      {Elastic page fill requires the balanced density}
      {Use ResumeDensity=balanced, or ResumePageFill=natural.}%
  }%
}{%
  \PackageError{resume-layout}
    {Unknown page fill mode '\ResumePageFill'}
    {Use natural or elastic. Falling back to natural.}%
  \renewcommand{\ResumePageFill}{natural}%
  \ResumeElasticPageFillfalse
}}

\newcommand{\ResumeInstallFiniteTextBottom}{%
  \def\@textbottom{\vskip\z@\@plus 10mm}%
  \let\@texttop\relax}
\makeatother

% In elastic mode, ship the final content page before LaTeX's end-document
% \newpage inserts an unbounded \vfil. Natural mode is intentionally a no-op.
\newcommand{\ResumeFinalizePage}{%
  \ifResumeElasticPageFill
    \par\penalty-10000\relax
  \fi}

\AtBeginDocument{%
  \fontsize{\ResumeBaseFontSize}{\ResumeBaseLineHeight}\selectfont
  \setlength{\parindent}{0pt}%
  \setlength{\parskip}{0pt plus 0.08pt}%
  \setlength{\emergencystretch}{1em}%
  \clubpenalty=10000
  \widowpenalty=10000
  \displaywidowpenalty=10000
  \brokenpenalty=8000
  \ifResumeFlushBottom
    \flushbottom
    \ifResumeElasticPageFill
      % \flushbottom resets \@textbottom, so install the finite buffer after it.
      \ResumeInstallFiniteTextBottom
    \fi
  \else
    \raggedbottom
  \fi
  \typeout{RESUME-LAYOUT profile=\ResumeDensity fill=\ResumePageFill}%
}
